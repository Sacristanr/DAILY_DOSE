add_shortcode('cds_area_cliente', 'cds_area_cliente_shortcode');

function cds_area_cliente_shortcode() {

    /* Evitar conflictos al editar con Elementor */
    if (
        class_exists('\Elementor\Plugin') &&
        isset(\Elementor\Plugin::$instance->editor) &&
        \Elementor\Plugin::$instance->editor->is_edit_mode()
    ) {
        return '<div class="cds-ficha-cliente"><p>Vista previa del área privada de clientes. El contenido real se muestra en frontend según el usuario que haya iniciado sesión.</p></div>';
    }

    if (!is_user_logged_in()) {
        ob_start();
        echo '<h3>Acceso</h3>';
        echo '<p>Por favor, introduce tus credenciales.</p>';

        if (isset($_GET['login']) && $_GET['login'] === 'failed') {
            echo '<p class="cds-login-error">ERROR: el nombre de usuario o la contraseña son incorrectos.</p>';
        }

        if (isset($_GET['login']) && $_GET['login'] === 'empty') {
            echo '<p class="cds-login-error">ERROR: debes introducir usuario y contraseña.</p>';
        }

        wp_login_form(array(
            'redirect' => home_url('/area-privada-cliente/'),
            'label_username' => 'Usuario',
            'label_password' => 'Contraseña',
            'label_remember' => 'Recuérdame',
            'label_log_in'   => 'Acceder'
        ));
        return ob_get_clean();
    }

    $user = wp_get_current_user();
    $roles = (array) $user->roles;

    if (in_array('soporte_cds', $roles) || in_array('administrator', $roles) || in_array('admin_cds', $roles)) {
        return '<p>Has iniciado sesión como usuario interno. Accede a tu panel correspondiente.</p><p><a class="cds-logout-btn" href="' . esc_url(wp_logout_url(home_url('/area-privada-cliente/'))) . '">Cerrar sesión</a></p>';
    }

    if (!in_array('cliente', $roles)) {
        return '<p>No tienes permisos para acceder a esta sección.</p>';
    }

    $user_id = get_current_user_id();

    $args = array(
        'post_type' => 'clientes',
        'posts_per_page' => 1,
        'post_status' => 'publish',
        'meta_query' => array(
            array(
                'key' => 'usuario_asociado',
                'value' => $user_id,
                'compare' => '='
            )
        )
    );

    $query = new WP_Query($args);

    if (!$query->have_posts()) {
        return '<p>No se ha encontrado una ficha asociada a tu usuario.</p><p><a class="cds-logout-btn" href="' . esc_url(wp_logout_url(home_url('/area-privada-cliente/'))) . '">Cerrar sesión</a></p>';
    }

    ob_start();

    while ($query->have_posts()) {
        $query->the_post();

        $codigo = get_field('codigo_cliente');
        $cif = get_field('cif');
        $contacto = get_field('contacto');
        $email = get_field('email_cliente');
        $telefono = get_field('telefono_cliente');
        $estado = get_field('estado_cliente');
        $servicios = get_field('servicios_contratados');
        $nivel = get_field('nivel_digitalizacion');
        $tecnico = get_field('tecnico_responsable');
        $observaciones = get_field('observaciones_internas');
        $logout_url = wp_logout_url(home_url('/area-privada-cliente/'));

        echo '<div class="cds-ficha-cliente">';
        echo '<h2>' . esc_html(get_the_title()) . '</h2>';
        echo '<p><strong>Código de cliente:</strong> ' . esc_html($codigo) . '</p>';
        echo '<p><strong>CIF / NIF:</strong> ' . esc_html($cif) . '</p>';
        echo '<p><strong>Persona de contacto:</strong> ' . esc_html($contacto) . '</p>';
        echo '<p><strong>Correo electrónico:</strong> ' . esc_html($email) . '</p>';
        echo '<p><strong>Teléfono:</strong> ' . esc_html($telefono) . '</p>';
        echo '<p><strong>Estado del cliente:</strong> ' . esc_html($estado) . '</p>';

        echo '<p><strong>Servicios contratados:</strong> ';
        if (is_array($servicios) && !empty($servicios)) {
            echo esc_html(implode(', ', $servicios));
        } else {
            echo 'Sin datos';
        }
        echo '</p>';

        echo '<p><strong>Nivel de digitalización:</strong> ' . esc_html($nivel) . '</p>';
        echo '<p><strong>Técnico responsable:</strong> ' . esc_html($tecnico) . '</p>';

        if (!empty($observaciones)) {
            echo '<div class="cds-observaciones">';
            echo '<strong>Observaciones:</strong>';
            echo '<div class="cds-observaciones-texto">';
            echo wpautop(wp_kses_post($observaciones));
            echo '</div>';
            echo '</div>';
        }

        $documentacion = get_field('documentacion_validada');
        $propuestas = get_field('propuestas_presupuestos');
        $facturacion = get_field('facturacion');
        $informes = get_field('informes_seguimiento');

        cds_mostrar_archivo_cliente($documentacion, 'Documentación validada');
        cds_mostrar_archivo_cliente($propuestas, 'Propuestas y presupuestos');
        cds_mostrar_archivo_cliente($facturacion, 'Facturación');
        cds_mostrar_archivo_cliente($informes, 'Informes y seguimiento');

        echo '</div>';

        echo '<div class="cds-ficha-acciones">';
        echo '<a class="cds-logout-btn" href="' . esc_url($logout_url) . '">Cerrar sesión</a>';
        echo '</div>';
    }

    wp_reset_postdata();

    return ob_get_clean();
}

function cds_mostrar_archivo_cliente($archivo, $titulo) {

    if (empty($archivo)) {
        return;
    }

    echo '<div class="cds-bloque-documental">';
    echo '<h3>' . esc_html($titulo) . '</h3>';

    if (is_array($archivo) && !empty($archivo['url'])) {
        $url = $archivo['url'];
        $nombre = !empty($archivo['title']) ? $archivo['title'] : basename($url);

        echo '<p><a href="' . esc_url($url) . '" target="_blank" rel="noopener noreferrer">' . esc_html($nombre) . '</a></p>';
    } elseif (is_string($archivo)) {
        $nombre = basename($archivo);

        echo '<p><a href="' . esc_url($archivo) . '" target="_blank" rel="noopener noreferrer">' . esc_html($nombre) . '</a></p>';
    } else {
        echo '<p>Archivo no disponible</p>';
    }

    echo '</div>';
}
