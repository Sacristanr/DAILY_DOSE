add_action('acf/save_post', 'cds_generar_codigo_cliente_automatico', 20);

function cds_generar_codigo_cliente_automatico($post_id) {
    // Evitar autosaves, opciones, revisiones, etc.
    if (!$post_id || !is_numeric($post_id)) {
        return;
    }

    if (get_post_type($post_id) !== 'clientes') {
        return;
    }

    if (wp_is_post_revision($post_id)) {
        return;
    }

    // Si ya tiene código, no lo vuelve a generar
    $codigo_actual = get_field('codigo_cliente', $post_id);
    if (!empty($codigo_actual)) {
        return;
    }

    // Recupera el contador actual
    $contador = get_option('cds_contador_clientes', 2);
    $contador++;
    
    // Formato que usamos --> CDS-0001
    $codigo_nuevo = 'CDS-' . str_pad($contador, 4, '0', STR_PAD_LEFT);

    // Guardar código en Advanced Custom Fields-ACF
    update_field('codigo_cliente', $codigo_nuevo, $post_id);

    // Actualizar contador global
    update_option('cds_contador_clientes', $contador);
}
