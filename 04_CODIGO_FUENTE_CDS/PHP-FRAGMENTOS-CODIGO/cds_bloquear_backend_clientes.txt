add_action('admin_init', 'cds_bloquear_backend_clientes');
function cds_bloquear_backend_clientes() {
    if (defined('DOING_AJAX') && DOING_AJAX) {
        return;
    }

    if (!is_user_logged_in()) {
        return;
    }

    $user = wp_get_current_user();

    if (in_array('cliente', (array) $user->roles)) {
        wp_redirect(home_url('/area-privada-cliente/'));
        exit;
    }
}
